# Nginx 配置 (WebSocket 转发)
location /ws {
    proxy_pass http://127.0.0.1:8080/ws;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "Upgrade";
    proxy_set_header Host $host;
}

location /api/ {
    proxy_pass http://127.0.0.1:8080/api/;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
}

# 增量 SQL (如需手动初始化)
-- 潜在用户表 (已在 database.py 自动创建)
CREATE TABLE IF NOT EXISTS potential_users (
    user_id INTEGER PRIMARY KEY,
    username TEXT,
    first_name TEXT,
    last_name TEXT,
    message_text TEXT,
    keywords TEXT,
    group_id INTEGER,
    group_title TEXT,
    captured_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 钱包监听表 (已在 database.py 自动创建)
CREATE TABLE IF NOT EXISTS address_monitors (
    monitor_id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER,
    address TEXT NOT NULL,
    network TEXT NOT NULL,
    last_tx_timestamp INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT 1,
    FOREIGN KEY (user_id) REFERENCES users (user_id)
);
